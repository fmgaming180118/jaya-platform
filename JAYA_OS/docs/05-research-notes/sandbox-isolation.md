# Sandbox Isolation — JAYA_OS

## Overview

JAYA_OS implements a **multi-layer sandbox** for all mounted features. The sandbox ensures that features (whether generated by the brain or user-installed) cannot access system resources beyond their declared permissions.

### Sandbox Execution Context
```python
def create_sandbox_namespace(allowed_imports, blocked_builtins, proxies):
    """Create a restricted namespace for feature execution."""
    namespace = {
        "__builtins__": create_restricted_builtins(blocked_builtins),
        "__import__": create_restricted_import(allowed_imports),
    }
    
    # Inject capability proxies
    namespace.update(proxies)
    
    # Add safe standard library modules
    for mod_name in allowed_imports:
        if mod_name in sys.modules:
            namespace[mod_name] = sys.modules[mod_name]
    
    return namespace

def create_restricted_builtins(blocked):
    """Create a restricted __builtins__ dict."""
    import builtins
    restricted = {}
    for name in dir(builtins):
        if name not in blocked:
            restricted[name] = getattr(builtins, name)
    return restricted

def create_restricted_import(allowed):
    """Create a restricted import function."""
    def restricted_import(name, *args, **kwargs):
        if name not in allowed:
            raise ImportError(f"Import not allowed: {name}")
        return __import__(name, *args, **kwargs)
    return restricted_import
```

## Layer 2: Capability Proxies

### Filesystem Proxy
```python
class FilesystemProxy:
    def __init__(self, allowed_read: List[str], allowed_write: List[str]):
        self.allowed_read = [Path(p).resolve() for p in allowed_read]
        self.allowed_write = [Path(p).resolve() for p in allowed_write]
    
    def _check_read(self, path: Path) -> bool:
        resolved = path.resolve()
        return any(resolved.is_relative_to(p) for p in self.allowed_read)
    
    def _check_write(self, path: Path) -> bool:
        resolved = path.resolve()
        return any(resolved.is_relative_to(p) for p in self.allowed_write)
    
    def read_text(self, path: str) -> str:
        path = Path(path)
        if not self._check_read(path):
            raise PermissionError(f"Read not allowed: {path}")
        return path.read_text()
    
    def write_text(self, path: str, content: str) -> None:
        path = Path(path)
        if not self._check_write(path):
            raise PermissionError(f"Write not allowed: {path}")
        path.write_text(content)
    
    def list_dir(self, path: str) -> List[str]:
        path = Path(path)
        if not self._check_read(path):
            raise PermissionError(f"List not allowed: {path}")
        return [p.name for p in path.iterdir()]
    
    def exists(self, path: str) -> bool:
        path = Path(path)
        if not self._check_read(path):
            raise PermissionError(f"Exists check not allowed: {path}")
        return path.exists()
    
    def is_file(self, path: str) -> bool:
        path = Path(path)
        if not self._check_read(path):
            raise PermissionError(f"Is file check not allowed: {path}")
        return path.is_file()
    
    def is_dir(self, path: str) -> bool:
        path = Path(path)
        if not self._check_read(path):
            raise PermissionError(f"Is dir check not allowed: {path}")
        return path.is_dir()
```

### Network Proxy
```python
class NetworkProxy:
    def __init__(self, allowed_domains: List[str], max_requests_per_minute: int = 10):
        self.allowed_domains = allowed_domains
        self.max_requests = max_requests_per_minute
        self.request_times = []
    
    def _check_domain(self, url: str) -> bool:
        parsed = urlparse(url)
        return any(parsed.netloc.endswith(d) for d in self.allowed_domains)
    
    def _check_rate_limit(self) -> bool:
        now = time.time()
        self.request_times = [t for t in self.request_times if now - t < 60]
        return len(self.request_times) < self.max_requests
    
    def request(self, method: str, url: str, **kwargs) -> requests.Response:
        if not self._check_domain(url):
            raise PermissionError(f"Domain not allowed: {urlparse(url).netloc}")
        if not self._check_rate_limit():
            raise PermissionError("Rate limit exceeded")
        
        self.request_times.append(time.time())
        return requests.request(method, url, **kwargs)
    
    def get(self, url: str, **kwargs) -> requests.Response:
        return self.request("GET", url, **kwargs)
    
    def post(self, url: str, **kwargs) -> requests.Response:
        return self.request("POST", url, **kwargs)
```

## Layer 3: Resource Limits

### Windows (Job Objects)
```python
import win32job
import win32api
import win32con

class WindowsResourceLimiter:
    def __init__(self, cpu_percent: int = 50, memory_mb: int = 200):
        self.job = win32job.CreateJobObject(None, "")
        
        # CPU limit (percentage * 10000)
        info = win32job.QueryInformationJobObject(
            self.job, win32job.JobObjectBasicLimitInformation
        )
        info['LimitFlags'] = win32job.JOB_OBJECT_LIMIT_PROCESS_TIME
        info['PerProcessUserTimeLimit'] = int(cpu_percent * 10000)
        win32job.SetInformationJobObject(
            self.job, win32job.JobObjectBasicLimitInformation, info
        )
        
        # Memory limit
        info = win32job.QueryInformationJobObject(
            self.job, win32job.JobObjectExtendedLimitInformation
        )
        info['LimitFlags'] |= win32job.JOB_OBJECT_LIMIT_PROCESS_MEMORY
        info['ProcessMemoryLimit'] = memory_mb * 1024 * 1024
        win32job.SetInformationJobObject(
            self.job, win32job.JobObjectExtendedLimitInformation, info
        )
    
    def assign_process(self, pid: int):
        handle = win32api.OpenProcess(win32con.PROCESS_ALL_ACCESS, False, pid)
        win32job.AssignProcessToJobObject(self.job, handle)
```

### Linux (cgroups v2)
```python
import subprocess
import os

class LinuxResourceLimiter:
    def __init__(self, cpu_percent: int = 50, memory_mb: int = 200):
        self.cgroup_path = f"/sys/fs/cgroup/jaya_os/feature_{os.getpid()}"
        os.makedirs(self.cgroup_path, exist_ok=True)
        
        # CPU limit (quota/period)
        with open(f"{self.cgroup_path}/cpu.max", "w") as f:
            f.write(f"{int(cpu_percent * 1000)} 100000")
        
        # Memory limit
        with open(f"{self.cgroup_path}/memory.max", "w") as f:
            f.write(f"{memory_mb * 1024 * 1024}")
    
    def assign_process(self, pid: int = None):
        pid = pid or os.getpid()
        with open(f"{self.cgroup_path}/cgroup.procs", "w") as f:
            f.write(str(pid))
```

## Layer 4: Process Isolation (Future)

### Multiprocessing Isolation
```python
from multiprocessing import Process, Queue
import sys

class FeatureProcess:
    def __init__(self, feature_module: str, feature_id: str):
        self.feature_module = feature_module
        self.feature_id = feature_id
        self.process = None
        self.request_queue = Queue()
        self.response_queue = Queue()
    
    def start(self):
        self.process = Process(
            target=self._run_feature,
            args=(self.feature_module, self.feature_id,
                  self.request_queue, self.response_queue),
            daemon=True
        )
        self.process.start()
    
    def _run_feature(self, module_name, feature_id, req_queue, resp_queue):
        # Set up sandbox in child process
        from src.os_kernel.sandbox import setup_sandbox
        setup_sandbox(feature_id)
        
        # Import and run feature
        module = __import__(module_name, fromlist=['handle_request'])
        while True:
            request = req_queue.get()
            if request is None:  # Shutdown signal
                break
            result = module.handle_request(request)
            resp_queue.put(result)
    
    def dispatch(self, action: str, payload: dict) -> Any:
        self.request_queue.put({"action": action, "payload": payload})
        return self.response_queue.get()
    
    def stop(self):
        self.request_queue.put(None)
        self.process.join(timeout=5)
        if self.process.is_alive():
            self.process.terminate()
```

## Violation Handling

| Violation | Detection | Response |
|---|---|---|
| Import not whitelisted | Import hook | `ImportError`, terminate feature |
| Blocked builtin called | Wrapper | `SecurityError`, terminate feature |
| FS access outside allowed paths | Proxy | `PermissionError`, log, terminate |
| Network to unallowed domain | Proxy | `PermissionError`, log, terminate |
| CPU limit exceeded | cgroups/Job | Throttle, then terminate |
| Memory limit exceeded | cgroups/Job | Terminate immediately |
| Execution timeout | Watchdog | `TimeoutError`, terminate |
